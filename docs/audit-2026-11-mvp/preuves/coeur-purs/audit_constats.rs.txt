fn panic_message(panic: &(dyn std::any::Any + Send)) -> String {
    panic
        .downcast_ref::<String>()
        .cloned()
        .or_else(|| panic.downcast_ref::<&str>().map(|s| (*s).to_owned()))
        .unwrap_or_default()
}

fn response(status: u16, body: &str) -> ResponseData {
    ResponseData {
        status,
        duration_ms: 1,
        headers: vec![("Content-Type".to_owned(), "application/json".to_owned())],
        body: body.to_owned(),
    }
}

/// Dossiers de premier niveau que la barre latérale montre.
fn listed(root: &Path) -> Vec<String> {
    crate::tree::read(root)
        .unwrap_or_default()
        .into_iter()
        .filter_map(|node| match node {
            TreeNode::Folder { path, .. } => Some(path),
            TreeNode::Request { .. } => None,
        })
        .collect()
}

/// Fichiers sous `path`, liens non suivis.
fn count_files(path: &Path) -> usize {
    let Ok(meta) = fs::symlink_metadata(path) else {
        return 0;
    };
    if meta.is_dir() {
        fs::read_dir(path)
            .into_iter()
            .flatten()
            .flatten()
            .map(|entry| count_files(&entry.path()))
            .sum()
    } else {
        usize::from(meta.is_file())
    }
}

/// `true` si `work` finit avant `limit`.
fn finishes_within(limit: Duration, work: impl FnOnce() + Send + 'static) -> bool {
    let (done, wait) = mpsc::channel();
    std::thread::spawn(move || {
        work();
        let _ = done.send(());
    });
    wait.recv_timeout(limit).is_ok()
}

/// Une expression JSONPath en cours de frappe fait paniquer le filtre.
#[test]
fn filtre_jsonpath_echappement_incomplet() {
    let body = r#"{"a": 1}"#;
    let mut paniques = Vec::new();
    for expression in ["$['\\u", "$[\"\\u", "$[?@.a=='\\u"] {
        if let Err(panic) =
            catch_unwind(AssertUnwindSafe(|| crate::response::filter(body, expression)))
        {
            eprintln!("{expression:?} -> PANIQUE : {}", panic_message(&*panic));
            paniques.push(expression);
        }
    }
    assert!(paniques.is_empty(), "expressions qui paniquent : {paniques:?}");
}

/// Un corps HTML fait de balises ouvertes jamais fermées donne un aperçu aussi profond que le
/// corps est long : la sérialisation (et la libération) récursives débordent la pile.
#[test]
#[ignore = "déborde la pile et arrête le processus de test : à lancer seul"]
fn apercu_html_imbrication_profonde() {
    // 1 Mo ; la limite d'un corps est de 5 Mo.
    let body = "<div>".repeat(200_000);
    let handle = std::thread::Builder::new()
        // Pile du fil principal sous Linux (1 Mo sous Windows).
        .stack_size(8 * 1024 * 1024)
        .spawn(move || {
            let nodes = crate::response::preview(&body);
            eprintln!("aperçu construit");
            let size = serde_json::to_string(&nodes).map(|text| text.len());
            eprintln!("aperçu sérialisé : {size:?}");
            drop(nodes);
            eprintln!("aperçu libéré");
        });
    assert!(handle.is_ok_and(|h| h.join().is_ok()));
}

/// La mise en forme indente chaque ligne de sa profondeur : des balises ouvertes sans
/// fermeture (`<li>`, `<p>`, `<td>` du HTML courant) coûtent le carré de leur nombre.
#[test]
fn mise_en_forme_html_indentation_quadratique() {
    let size = |items: usize| -> (usize, usize) {
        let body = "<li>x".repeat(items);
        let rendered = crate::response::render(
            &body,
            &[],
            Some(crate::response::ResponseFormat::Html),
        );
        let bytes: usize = rendered
            .lines
            .iter()
            .flatten()
            .map(|token| token.text.len())
            .sum();
        (body.len(), bytes)
    };
    let (small_body, small) = size(1000);
    let (body, bytes) = size(4000);
    eprintln!("1000 <li> : corps {small_body} octets, jetons {small} octets");
    eprintln!("4000 <li> : corps {body} octets, jetons {bytes} octets");
    assert!(
        bytes < body * 50,
        "4000 <li> sans </li> ({body} octets) donnent {bytes} octets de jetons ; à 200 000 (1 Mo) : {} Go",
        2 * 200_000usize * 200_000 / 1_000_000_000
    );
}

/// Un contrat reçu d'un collègue peut figer Arnod : `minLength` très grand (exemple construit
/// caractère par caractère), ou `anyOf` qui se référence deux fois (2^64 essais).
#[test]
fn contrat_qui_fige_arnod() {
    let sample = finishes_within(Duration::from_secs(5), || {
        let contract = crate::contract::parse::load(
            r##"openapi: 3.0.3
info: { title: T, version: "1" }
paths:
  /a:
    post:
      requestBody:
        content:
          application/json:
            schema: { type: object, properties: { note: { type: string, minLength: 4000000 } } }
      responses: { "200": { description: ok } }
"##,
        )
        .expect("contrat lisible");
        let _ = crate::contract::generate::request_for(
            &contract,
            &contract.operations[0],
            "contracts/t.yaml",
        );
    });
    let conformance = finishes_within(Duration::from_secs(5), || {
        let contract = crate::contract::parse::load(
            r##"openapi: 3.0.3
info: { title: T, version: "1" }
paths:
  /a:
    get:
      responses:
        "200":
          description: ok
          content:
            application/json:
              schema: { $ref: "#/components/schemas/Boucle" }
components:
  schemas:
    Boucle:
      anyOf:
        - { $ref: "#/components/schemas/Boucle" }
        - { $ref: "#/components/schemas/Boucle" }
"##,
        )
        .expect("contrat lisible");
        let _ = crate::contract::conformance::check(&contract, "GET /a", &response(200, "{}"));
    });
    assert!(
        sample && conformance,
        "en 5 s : requête créée (minLength 4 000 000) finie = {sample}, conformité (anyOf en boucle) finie = {conformance}"
    );
}

/// `decode_entities` cherche le `;` jusqu'à la fin du texte pour chaque `&` : quadratique.
#[test]
fn entites_html_quadratiques() {
    let time = |count: usize| {
        let text = "&a=1 ".repeat(count);
        let start = Instant::now();
        let _ = crate::response::markup::decode_entities(&text);
        start.elapsed()
    };
    let small = time(40_000);
    let large = time(160_000);
    eprintln!("200 Ko : {small:?} ; 800 Ko : {large:?}");
    assert!(
        large < small * 8,
        "texte 4 fois plus long, {:.0} fois plus lent (200 Ko : {small:?}, 800 Ko : {large:?})",
        large.as_secs_f64() / small.as_secs_f64()
    );
}

/// Renommer ou déplacer un dossier vers un nom réservé ou caché le fait disparaître.
#[test]
fn dossier_renomme_en_nom_reserve_disparait() {
    let dir = tempfile::tempdir().unwrap();
    let root = dir.path();
    let mut gaps = Vec::new();
    for (index, name) in ["contracts", "environments", ".archive"].iter().enumerate() {
        let folder = crate::tree::create_folder(root, "", &format!("Dossier {index}")).unwrap();
        crate::tree::create_request(root, &folder, Some("Requête")).unwrap();
        let renamed = crate::tree::rename(root, &folder, name);
        let shown = listed(root);
        if let Ok(path) = &renamed
            && !shown.contains(path)
        {
            gaps.push(format!("renommé en « {name} » : accepté, absent de l'arborescence"));
        }
        // Le dossier est remis à l'écart pour le cas suivant.
        let _ = fs::rename(root.join(name), root.join(format!("fait-{index}")));
    }
    let parent = crate::tree::create_folder(root, "", "Parent").unwrap();
    let inner = crate::tree::create_folder(root, &parent, "contracts").unwrap();
    crate::tree::create_request(root, &inner, Some("Requête")).unwrap();
    if let Ok(path) = crate::tree::move_to(root, &inner, "")
        && !listed(root).contains(&path)
    {
        gaps.push("« Parent/contracts » déplacé à la racine : accepté, absent de l'arborescence".to_owned());
    }
    assert!(gaps.is_empty(), "{gaps:#?}");
}

/// « Annuler » après une suppression définitive groupée ne remet que les fichiers texte.
#[test]
fn annuler_une_suppression_groupee_perd_des_fichiers() {
    use crate::tree::batch::{BatchAction, apply};
    let dir = tempfile::tempdir().unwrap();
    let root = dir.path();
    let folder = crate::tree::create_folder(root, "", "Commandes").unwrap();
    crate::tree::create_request(root, &folder, Some("Requête")).unwrap();
    // Pièce jointe binaire, requête d'un collègue en ISO-8859-1, sous-dossier encore vide.
    fs::write(root.join("Commandes/logo.png"), [0x89, b'P', b'N', b'G', 0xff, 0x00]).unwrap();
    fs::write(root.join("Commandes/ancien.yml"), b"info:\n  name: Cr\xe9er\n").unwrap();
    fs::create_dir(root.join("Commandes/À faire")).unwrap();
    let (_, undo) = apply(
        root,
        &[folder],
        &BatchAction::Delete { permanent: true },
    )
    .unwrap();
    undo.apply().unwrap();
    let missing: Vec<&str> = ["requete.yml", "logo.png", "ancien.yml", "À faire"]
        .into_iter()
        .filter(|name| !root.join("Commandes").join(name).exists())
        .collect();
    assert!(missing.is_empty(), "non remis par « Annuler » : {missing:?}");
}

/// Un lien vers le dossier lui-même (cas R2 de l'audit précédent, corrigé pour la lecture) :
/// la suppression groupée lit le dossier 41 fois et « Annuler » écrit 41 copies ; la
/// duplication échoue. Deux liens : 2^40.
#[cfg(unix)]
#[test]
fn lien_en_boucle_dans_une_action_groupee() {
    use crate::tree::batch::{BatchAction, apply};
    let dir = tempfile::tempdir().unwrap();
    let root = dir.path();
    let folder = crate::tree::create_folder(root, "", "Commandes").unwrap();
    crate::tree::create_request(root, &folder, Some("Requête")).unwrap();
    std::os::unix::fs::symlink(".", root.join("Commandes/boucle")).unwrap();
    let duplicated = apply(root, std::slice::from_ref(&folder), &BatchAction::Duplicate)
        .map(|(done, _)| done.items)
        .map_err(|e| e.message);
    eprintln!("dupliquer : {duplicated:?}");
    let before = count_files(&root.join("Commandes"));
    let (_, undo) = apply(root, &[folder], &BatchAction::Delete { permanent: true }).unwrap();
    undo.apply().unwrap();
    let after = count_files(&root.join("Commandes"));
    assert!(
        duplicated.is_ok() && after == before,
        "dupliquer : {duplicated:?} ; fichiers avant suppression : {before}, après « Annuler » : {after}"
    );
}

/// Assertions Bruno dont l'expression n'est pas un simple chemin (`.length`) : reprises sans
/// avertissement, elles passent à tort (`neq`) ou échouent à tort (`gt`).
#[test]
fn assertion_bruno_sur_une_expression() {
    use crate::bruno::read::{Collection, Folder, Source};
    use crate::bruno::{ItemKind, ItemStatus};
    let file = crate::bruno::parse::parse(
        "meta {\n  name: Lister\n  type: http\n}\n\nget {\n  url: https://api.test/items\n}\n\nassert {\n  res.body.items.length: neq 0\n  res.body.items.length: gt 0\n}\n",
    )
    .unwrap();
    let collection = Collection {
        name: "C".to_owned(),
        tree: Folder {
            requests: vec![Source {
                path: "Lister.bru".to_owned(),
                file,
            }],
            ..Folder::default()
        },
        bru_files: 1,
        ..Collection::default()
    };
    let converted = crate::bruno::convert::convert(&collection);
    let request =
        crate::opencollection::request::from_value(&converted.files[0].value, "x").unwrap();
    let passed = |body: &str| -> Vec<bool> {
        crate::request::assertions::evaluate_all(&request.assertions, &response(200, body))
            .iter()
            .map(|r| r.passed)
            .collect()
    };
    let warned = converted.items.iter().any(|item| {
        item.kind == ItemKind::Assertion
            && matches!(item.status, ItemStatus::ToCheck | ItemStatus::Skipped)
    });
    let empty = passed(r#"{"items":[]}"#);
    let full = passed(r#"{"items":[1,2]}"#);
    assert!(
        warned || (empty == [false, false] && full == [true, true]),
        "signalé dans le rapport : {warned} ; liste vide [neq 0, gt 0] = {empty:?} (attendu [false, false]) ; 2 éléments = {full:?} (attendu [true, true])"
    );
}

        }
    }
    let name = crate::contract::generate::safe_name(" .. .. ", "Autres");
    if name == ".." || name.ends_with('.') {
        gaps.push(format!("étiquette « .. .. » → dossier {name:?}"));
    }
    let value: Value = json!(crate::request::dynamic::draw(
        "$date:+99999999999d",
        &mut crate::request::dynamic::Dice::seeded(7, 1_794_495_731_000)
    ));
    eprintln!("$date:+99999999999d -> {value}");
    assert!(gaps.is_empty(), "{gaps:#?}");
}
